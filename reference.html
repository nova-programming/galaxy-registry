Type System

Nova is statically typed with inference. All types resolved at compile time.

TypeSizeDescriptionExample Literal
int4 bytes32-bit signed integer42, 0xFF (hex), 0b1010 (binary)
float8 bytes64-bit IEEE-7543.14, -0.5
stringheapImmutable UTF-8 text"hello"
bool1 byteBooleantrue, false
listheapDynamic array of type T[1, 2, 3]
dataheapStruct with named fieldsdata Point { x: int, y: int }
Type inference Omit the type annotation and Nova will infer it: pi = 3.14 infers float, count = 10 infers int.

Enum

Named integer constants with auto-incrementing values, desugared to const assignments at compile time.

enum Color { Red, Green, Blue }
// Desugars to:
// const Color_Red = 0
// const Color_Green = 1
// const Color_Blue = 2

HashMap / Dictionary

Key-value store with {"key": value} literal syntax. Keys must be strings; values can be any type.

d = {"name": "Alice", "age": 10}
print(d.get("name"))      # Alice
print(d.get("missing"))   # 0 (default)
d.set("name", "Bob")      # update value
print(d.has("name"))      # True
print(len(d))             # 2
d.remove("name")          # delete key
print(d.keys())           # list of keys
print(d.values())         # list of values
print(d.items())          # [key1, val1, key2, val2]

Control Flow

If / Else If / Else

if score >= 90 {
    print("A")
} else if score >= 80 {
    print("B")
} else {
    print("C")
}

While

i = 0
while i < 10 {
    print(i)
    i = i + 1
}

For Loop

# Ascending
for i = 1 to 10 step 1 {
    print(i)
}

# Descending
for i = 10 downto 1 step 1 {
    print(i)
}

# Custom step
for i = 0 to 100 step 10 {
    print(i)
}

Switch

Multi-branch conditional that desugars to if-elif-else. Works with any comparable type (int, string, bool).

switch x {
    case 1 { print("one") }
    case 2 { print("two") }
    else { print("other") }
}

Exceptions

Nova supports structured exception handling via try/catch/throw, using setjmp/longjmp in native compilation.

try {
    throw "something went wrong"
} catch(e) {
    print(e.message)
}

# No exception: catch body skipped
try {
    print("all good")
} catch(e) {
    print("not reached")
}

# throw inside function
def safe_div(a, b) {
    if b == 0 {
        throw "divide by zero"
    }
    print(a / b)
}

try {
    safe_div(10, 0)
} catch(e) {
    print(-1)
}

List Comprehensions

Syntax: [expr for x in list if cond]. Desugars at parse time to a ForIn loop with append. The if clause is optional.

# Basic
items = [1, 2, 3, 4, 5]
result = [x * 2 for x in items]

# With filter
evens = [x for x in items if x % 2 == 0]

# Inline
print([x * 10 for x in items if x > 2])

Functions

# With return type annotation
def add(a: int, b: int) -> int {
    return a + b
}

# Inferred return type (no -> annotation)
def greet(name: string) {
    print("Hello, " + name)
}

# Recursion
def factorial(n: int) -> int {
    if n <= 1 {
        return 1
    }
    return n * factorial(n - 1)
}

# Call top-level code
print(add(3, 4))

Classes

class Player {
    health: int
    name: string

    def __init__(h: int, n: string) {
        self.health = h
        self.name = n
    }

    def take_damage(amount: int) {
        self.health = self.health - amount
    }

    def is_alive() -> bool {
        return self.health > 0
    }
}

# Usage
p = Player(100, "Hero")
p.take_damage(25)
print(p.is_alive())   # true

Inline Assembly

# Insert raw assembly instructions
@raw {
    mov eax, 42
    ret
}

# Access Nova variables by name
x = 10
@raw {
    mov eax, [x]    # load x into eax
    add eax, 1
    mov [x], eax    # store back
}

Grammar (EBNF)

program         = { top_level }
top_level       = function_def | class_def | data_def | enum_def | import_stmt | statement

function_def    = "def" id "(" [ params ] ")" [ "->" type ] block
class_def       = "class" id "{" { field | method } "}"
data_def        = "data" id "{" { field } "}"
enum_def        = "enum" id "{" id { "," id } "}"
import_stmt     = "import" id [ "as" id ]

block           = "{" { statement } "}"
statement       = if_stmt | switch_stmt | while_stmt | for_stmt | for_in_stmt
                | return_stmt | break_stmt | continue_stmt
                | const_stmt | assign_stmt | call_stmt | print_stmt

if_stmt         = "if" expr block { "elif" expr block } [ "else" block ]
switch_stmt     = "switch" expr "{" { "case" expr block } [ "else" block ] "}"
while_stmt      = "while" expr block
for_stmt        = "for" id "=" expr ( "to" | "downto" ) expr "step" expr block
for_in_stmt     = "for" id "in" expr block
list_comp       = "[" expr "for" id "in" expr [ "if" expr ] "]"
try_stmt        = "try" block "catch" id block
throw_stmt      = "throw" expr
return_stmt     = "return" [ expr ]
break_stmt      = "break"
continue_stmt   = "continue"
const_stmt      = "const" id "=" expr

assign_stmt     = id "=" expr
                | id "[" expr "]" "=" expr

expr            = or_expr
or_expr         = and_expr { "or" and_expr }
and_expr        = bit_or_expr { "and" bit_or_expr }
bit_or_expr     = bit_xor_expr { "|" bit_xor_expr }
bit_xor_expr    = bit_and_expr { "^" bit_and_expr }
bit_and_expr    = shift_expr { "&" shift_expr }
shift_expr      = comp_expr { ( "<<" | ">>" ) comp_expr }
comp_expr       = add_expr [ ( "==" | "!=" | "<" | ">" | "<=" | ">=" ) add_expr ]
add_expr        = mul_expr { ( "+" | "-" ) mul_expr }
mul_expr        = unary_expr { ( "*" | "/" | "%" ) unary_expr }
unary_expr      = ( "not" | "~" | "-" ) unary_expr | postfix_expr
postfix_expr    = primary { "[" expr "]" | "." id | "^" }
primary         = int_literal | float_literal | string_literal
                | bool_literal | id | "(" expr ")"
                | id "(" [ args ] ")"      # function call
                | "[" [ expr { "," expr } ] "]"       # list literal
                | "{" [ expr ":" expr { "," expr ":" expr } ] "}"   # dict literal

string_literal  = '"' { char | interpolation } '"'
interpolation   = "{" expr "}"
int_literal     = decimal | "0x" hex_digit+ | "0b" binary_digit+

type            = "int" | "float" | "string" | "bool" | "list" | id
