Variables & Types

Demonstrates explicit types and type inference.
# Explicit type annotation
name: string = "Nova"
age: int = 42

# Type inference — Nova figures it out
pi = 3.14          # inferred as float
is_ready = true    # inferred as bool
count = 100        # inferred as int

# Constants
const MAX_SIZE = 4096

print(name)
print(age)
print(pi)
print(is_ready)
print(MAX_SIZE)
Output
Nova
42
3.14
true
4096

FizzBuzz

Classic programming problem — conditionals, loops, and modulo.
for i = 1 to 20 step 1 {
    if i % 15 == 0 {
        print("FizzBuzz")
    } else if i % 3 == 0 {
        print("Fizz")
    } else if i % 5 == 0 {
        print("Buzz")
    } else {
        print(i)
    }
}
Output
1 2 Fizz 4 Buzz Fizz 7 8 Fizz Buzz 11 Fizz 13 14 FizzBuzz 16 17 Fizz 19 Buzz

Sum to N

Loop from 1 to N and accumulate the total.
def sum_to(n) {
    s = 0
    i = 0
    while i <= n {
        s = s + i
        i = i + 1
    }
    return s
}

print(sum_to(100))
Output
5050

Prime Checker

Counts primes up to a limit using trial division.
def is_prime(n) {
    if n < 2 { return 0 }
    i = 2
    while i * i <= n {
        if n % i == 0 { return 0 }
        i = i + 1
    }
    return 1
}

def count_primes(limit) {
    count = 0
    i = 2
    while i <= limit {
        if is_prime(i) == 1 { count = count + 1 }
        i = i + 1
    }
    return count
}

print(count_primes(100))
Output
25

Fibonacci

Recursive Fibonacci — tests function call overhead and recursion depth.
def fib(n) {
    if n <= 1 { return n }
    return fib(n - 1) + fib(n - 2)
}

print(fib(10))
print(fib(20))
print(fib(35))
Output
55
6765
9227465

Random Numbers

Cryptographically secure random numbers via ChaCha20 PRNG (native only).
# Seed the ChaCha20 generator
chacha20_init(12345, 67890)

# Generate 5 random numbers
i = 0
while i < 5 {
    r = random()
    print("Random: " + str(r))
    i = i + 1
}
Output (example — values vary by seed)
Random: 123456789
Random: 987654321
Random: 555111222
Random: 333444777
Random: 888999000
Native only ChaCha20 functions (chacha20_init, chacha20_block) and random() are only available in compiled executables via nova build, not in the VM (nova dev).

Classes

Object-oriented programming with constructors, fields, and methods.
class Player {
    health: int
    name: string

    def __init__(h: int, n: string) {
        self.health = h
        self.name = n
    }

    def take_damage(amount: int) {
        self.health = self.health - amount
    }

    def is_alive() -> bool {
        return self.health > 0
    }
}

p = Player(100, "Hero")
p.take_damage(25)
print(p.is_alive())
print(p.health)
Output
true
75

String Interpolation

Embed expressions directly in strings with {expr} syntax.
name = "World"
count = 42
print("Hello {name}, count is {count}")
print("Sum: {10 + 20}")

// Output:
// Hello World, count is 42
// Sum: 30
Output
Hello World, count is 42
Sum: 30

def main() -> int {

Switch / Match

Multi-branch conditional that compiles to efficient if-elif-else chains.
def describe(x: int) -> string {
    switch x {
        case 1 { return "one" }
        case 2 { return "two" }
        else { return "many" }
    }
}

print(describe(1))  // one
print(describe(3))  // many
Output
one
many

Dictionary / HashMap

Key-value store with literal syntax and built-in methods.
def main() -> int {
    config = {"host": "localhost", "port": 8080}
    print(config.get("host"))
  
    config.set("host", "0.0.0.0")
    print(config.get("host"))
  
    print(len(config))
    print(config.has("port"))
    return 0
}
Output
localhost
0.0.0.0
2
true
